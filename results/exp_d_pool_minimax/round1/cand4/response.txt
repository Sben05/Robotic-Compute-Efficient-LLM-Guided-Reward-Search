```python
import torch
from typing import Dict, Tuple


@torch.jit.script
def compute_reward(
    root_states: torch.Tensor,
    actions: torch.Tensor,
    up_vec: torch.Tensor,
    dof_vel: torch.Tensor
) -> Tuple[torch.Tensor, Dict[str, torch.Tensor]]:
    # Root state layout: [xyz, quat(4), linear_vel(3), angular_vel(3)]
    # The ant runs toward the world +X direction.
    velocity = root_states[:, 7:10]
    forward_speed = velocity[:, 0]

    # Exponential shaping keeps a dense, strong gradient towards larger speeds.
    # The extra linear term continues rewarding speed beyond the exponential range.
    forward_speed_temperature: float = 4.0
    bounded_forward_speed = torch.clamp(forward_speed, min=0.0, max=10.0)
    positive_forward_speed = torch.clamp(forward_speed, min=0.0)
    forward_reward = (
        torch.exp(bounded_forward_speed / forward_speed_temperature) - 1.0
    ) + 0.2 * positive_forward_speed

    # Only an upright ant that is off the ground receives locomotion rewards.
    height = root_states[:, 2]
    alive = (height > 0.25).to(velocity.dtype)
    upright = torch.clamp(up_vec[:, 2], min=0.0, max=1.0)
    stable = alive * upright

    forward_reward = forward_reward * stable
    upright_bonus = 0.2 * upright * alive
    fall_penalty = -1.0 * (1.0 - stable)

    # Small regularizers discourage wasteful, jerky, or sideways motion.
    action_cost = 0.005 * torch.sum(actions * actions, dim=-1)
    lateral_velocity_cost = 0.01 * torch.sum(
        velocity[:, 1:3] * velocity[:, 1:3], dim=-1
    )
    joint_velocity_cost = 1e-5 * torch.sum(dof_vel * dof_vel, dim=-1)

    total_reward = (
        forward_reward
        + upright_bonus
        + fall_penalty
        - action_cost
        - lateral_velocity_cost
        - joint_velocity_cost
    )

    components = {
        "forward_speed": forward_reward,
        "upright_bonus": upright_bonus,
        "fall_penalty": fall_penalty,
        "action_cost": action_cost,
        "lateral_velocity_cost": lateral_velocity_cost,
        "joint_velocity_cost": joint_velocity_cost,
    }

    return total_reward, components
```
