from typing import Dict, Tuple
import torch


@torch.jit.script
def compute_reward(root_states: torch.Tensor,
                   targets: torch.Tensor,
                   actions: torch.Tensor,
                   dof_vel: torch.Tensor,
                   dt: float) -> Tuple[torch.Tensor, Dict[str, torch.Tensor]]:
    """
    Reward for making the ant run forward (towards its target) as fast as possible.

    Components
    ----------
    forward_speed : tanh-saturated forward velocity (m/s) along the heading-to-target
                    direction.  Dominant term, symmetric so that running backwards
                    is explicitly discouraged.
    upright      : world-z component of the torso up axis, keeps the ant on its legs
                    instead of flipping (which would otherwise be a cheap way to
                    "move" fast).
    lateral_penalty : squared side-slip velocity (keeps the run straight).
    spin_penalty : squared torso angular velocity (discourages tumbling/spinning).
    action_cost : mean squared joint action (control effort / smoothness).
    dof_vel_cost : mean squared joint velocity (penalty for jerky, high-frequency motion).
    """

    # ------------------------- configuration -------------------------
    speed_temperature = 2.0        # m/s  : softness of the forward-speed saturation
    speed_weight = 1.0             # weight of the forward-speed term
    upright_weight = 0.15          # weight of the stability term
    lateral_cost_scale = 0.05      # penalty on side-slip speed
    spin_cost_scale = 0.002        # penalty on torso angular velocity
    action_cost_scale = 0.01       # penalty on control effort
    dof_vel_cost_scale = 1.0e-4    # penalty on joint velocity

    # ------------------------- state extraction ----------------------
    pos = root_states[:, 0:3]
    lin_vel = root_states[:, 7:10]
    ang_vel = root_states[:, 10:13]
    quat = root_states[:, 3:7]     # (w, x, y, z)

    # desired forward direction: horizontal unit vector pointing at the target
    to_target = targets - pos
    to_target_h = torch.stack((to_target[:, 0], to_target[:, 1]), dim=1)
    target_dist = torch.norm(to_target_h, dim=1)
    forward = to_target_h / (target_dist.unsqueeze(1) + 1.0e-6)

    # horizontal velocity expressed in the (forward, lateral) frame
    vel_h = torch.stack((lin_vel[:, 0], lin_vel[:, 1]), dim=1)
    forward_speed = torch.sum(vel_h * forward, dim=1)
    lateral_speed = vel_h[:, 0] * forward[:, 1] - vel_h[:, 1] * forward[:, 0]

    # world-z component of the torso "up" axis (from the root quaternion)
    qx = quat[:, 1]
    qy = quat[:, 2]
    up_z = 1.0 - 2.0 * (qx * qx + qy * qy)

    # ------------------------- reward components ----------------------
    forward_speed_reward = torch.tanh(forward_speed / speed_temperature)
    upright_reward = torch.clamp(up_z, min=0.0)

    lateral_penalty = lateral_speed * lateral_speed
    spin_penalty = torch.mean(ang_vel * ang_vel, dim=1)
    action_penalty = torch.mean(actions * actions, dim=1)
    dof_vel_penalty = torch.mean(dof_vel * dof_vel, dim=1)

    # ------------------------- total reward --------------------------
    total_reward = (speed_weight * forward_speed_reward
                    + upright_weight * upright_reward
                    - lateral_cost_scale * lateral_penalty
                    - spin_cost_scale * spin_penalty
                    - action_cost_scale * action_penalty
                    - dof_vel_cost_scale * dof_vel_penalty)

    components: Dict[str, torch.Tensor] = {
        "forward_speed": forward_speed_reward,
        "upright": upright_reward,
        "lateral_penalty": lateral_penalty,
        "spin_penalty": spin_penalty,
        "action_cost": action_penalty,
        "dof_vel_cost": dof_vel_penalty,
    }

    return total_reward, components
