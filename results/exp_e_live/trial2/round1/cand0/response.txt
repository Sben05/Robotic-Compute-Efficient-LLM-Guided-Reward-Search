from typing import Dict, Tuple

import torch


@torch.jit.script
def compute_reward(root_states: torch.Tensor, actions: torch.Tensor, heading_vec: torch.Tensor,
                   up_vec: torch.Tensor) -> Tuple[torch.Tensor, Dict[str, torch.Tensor]]:
    """
    Reward for making the Ant run forward as fast as possible.

    root_states : (N, 13) -> [pos(3), quat(4, wxyz), lin_vel(3), ang_vel(3)]
    actions     : (N, A)   -> joint actions in [-1, 1]
    heading_vec : (N, 3)   -> world frame direction the ant is facing (its "forward")
    up_vec      : (N, 3)   -> world frame up axis of the ant's torso (z is up, index 2)
    """

    # ----------------------------- hyper parameters -----------------------------
    max_speed = 8.0             # [m/s] reference speed used for clipping / normalisation
    target_speed = 4.0          # [m/s] speed the ant is smoothly encouraged to reach
    speed_temperature = 1.5     # [m/s] temperature of the gaussian speed shaping
    upright_temperature = 2.0   # temperature of the exponential uprightness bonus

    w_speed = 1.0               # weight on forward (heading aligned) speed
    w_target = 0.5              # weight on the gaussian "reach a fast speed" shaping
    w_world = 0.2               # weight on the world +x speed (keeps running straight ahead)
    w_alive = 0.5               # weight on staying upright
    w_action = 0.05             # energy / smoothness penalty
    w_lateral = 0.1             # side slip penalty
    w_vertical = 0.05           # bouncing penalty
    w_angvel = 0.02             # body spin penalty

    # ----------------------------- forward direction -----------------------------
    heading = heading_vec.clone()
    heading[:, 2] = 0.0
    heading_norm = torch.norm(heading, dim=-1, keepdim=True)
    fallback = torch.zeros_like(heading)
    fallback[:, 0] = 1.0
    heading = torch.where(heading_norm > 1e-3, heading / (heading_norm + 1e-6), fallback)

    # ----------------------------- velocities -----------------------------------
    lin_vel = root_states[:, 7:10]
    ang_vel = root_states[:, 10:13]

    vel_xy = lin_vel.clone()
    vel_xy[:, 2] = 0.0

    forward_speed = torch.sum(vel_xy * heading, dim=-1)                      # [m/s]
    lateral_speed = vel_xy[:, 0] * (-heading[:, 1]) + vel_xy[:, 1] * heading[:, 0]
    world_speed = vel_xy[:, 0]                                              # +x speed [m/s]

    # ----------------------------- posture --------------------------------------
    up_z = torch.clamp(up_vec[:, 2], min=-1.0, max=1.0)
    upright_bonus = torch.exp(upright_temperature * (up_z - 1.0))            # in (0, 1]

    # ----------------------------- reward terms ---------------------------------
    # main term: velocity along the direction the ant faces
    speed_term = torch.clamp(forward_speed, min=-1.0, max=max_speed)
    # smooth bonus for approaching a high running speed
    target_term = torch.exp(-torch.pow(forward_speed - target_speed, 2) /
                            (2.0 * speed_temperature * speed_temperature))
    # keep the world +x progress (running straight ahead)
    world_term = torch.clamp(world_speed, min=-1.0, max=max_speed)
    # staying on its feet
    alive_term = upright_bonus
    # smoothness / energy
    action_term = torch.mean(torch.square(actions), dim=-1)
    lateral_term = torch.clamp(torch.abs(lateral_speed), min=0.0, max=max_speed) / max_speed
    vertical_term = torch.clamp(torch.abs(lin_vel[:, 2]), min=0.0, max=max_speed) / max_speed
    angvel_term = torch.clamp(torch.mean(torch.square(ang_vel), dim=-1), min=0.0, max=25.0) / 25.0

    reward = (w_speed * speed_term
              + w_target * target_term
              + w_world * world_term
              + w_alive * alive_term
              - w_action * action_term
              - w_lateral * lateral_term
              - w_vertical * vertical_term
              - w_angvel * angvel_term)

    components = {
        "forward_speed": speed_term,
        "target_speed_shaping": target_term,
        "world_x_speed": world_term,
        "upright_bonus": alive_term,
        "action_penalty": action_term,
        "lateral_penalty": lateral_term,
        "vertical_velocity_penalty": vertical_term,
        "angular_velocity_penalty": angvel_term,
    }
    return reward, components
