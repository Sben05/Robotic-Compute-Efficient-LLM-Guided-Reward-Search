import torch
from typing import Dict, Tuple

@torch.jit.script
def compute_reward(root_states: torch.Tensor, actions: torch.Tensor) -> Tuple[torch.Tensor, Dict[str, torch.Tensor]]:
    """
    Reward function for the task: "make the Ant run forward as fast as possible".

    Inputs (all taken from the environment attributes):
      root_states : (N, 13) tensor -> [pos(3), quat(w,x,y,z)(4), lin_vel(3), ang_vel(3)]
      actions     : (N, A) tensor -> the joint actions applied during the step

    Design:
      * main term   : forward (world +X / body-forward) speed, tanh-normalised so that the
                      reward stays bounded while keeping a useful gradient at high speeds.
      * high speed  : a second, much slower tanh term (nearly linear) that keeps rewarding
                      additional velocity beyond ~4 m/s.
      * alive       : upright bonus (body +Z axis aligned with world +Z) -> discourages
                      falling / tumbling in exchange for a short lived burst of speed.
      * ctrl cost   : small penalty on the action magnitude (smooth, energy friendly gait).
      * ang vel     : small penalty on large body angular velocities (no flipping).
      * fall        : penalty when the torso sinks close to the ground.
    """

    # ---------------- named hyper parameters (not inputs) ----------------
    speed_temperature: float = 3.0          # temperature of the (saturating) speed term
    high_speed_temperature: float = 15.0    # temperature of the slow / almost linear speed term
    angvel_temperature: float = 20.0        # temperature of the angular velocity penalty
    w_speed: float = 2.0
    w_high_speed: float = 0.5
    w_alive: float = 0.5
    w_ctrl: float = 0.05
    w_angvel: float = 0.2
    w_fall: float = 0.5

    # ---------------- body frame axes expressed in world frame ----------------
    # Isaac Gym stores the root quaternion as (w, x, y, z)
    q = root_states[:, 3:7]
    qw = q[:, 0]
    qx = q[:, 1]
    qy = q[:, 2]
    qz = q[:, 3]

    # body +X axis (the "nose" of the ant) in world coordinates
    fwd_x = 1.0 - 2.0 * (qy * qy + qz * qz)
    fwd_y = 2.0 * (qw * qz + qx * qy)
    # body +Z axis (up) -> z component in world coordinates (1 when perfectly upright)
    up_z = 1.0 - 2.0 * (qx * qx + qy * qy)

    # ---------------- kinematics ----------------
    lin_vel = root_states[:, 7:10]
    ang_vel = root_states[:, 10:13]
    world_speed_x = lin_vel[:, 0]

    # horizontal projection of the body forward axis (the world up axis is +Z)
    zero = torch.zeros_like(fwd_x)
    fwd_h = torch.stack([fwd_x, fwd_y, zero], dim=-1)
    fwd_h = fwd_h / (torch.norm(fwd_h, dim=-1, keepdim=True) + 1e-6)
    lin_vel_h = torch.stack([lin_vel[:, 0], lin_vel[:, 1], zero], dim=-1)
    body_speed = torch.sum(lin_vel_h * fwd_h, dim=-1)

    # blend of "running towards world +X" and "running in the direction the body faces":
    # running backwards is not rewarded, running forward (facing +X) is rewarded the most.
    forward_speed = 0.6 * world_speed_x + 0.4 * body_speed

    # ---------------- reward components ----------------
    speed_component = torch.tanh(forward_speed / speed_temperature)
    high_speed_component = torch.tanh(forward_speed / high_speed_temperature)
    alive_component = torch.clamp(up_z, min=0.0, max=1.0)
    ctrl_cost = torch.mean(actions * actions, dim=-1)
    ang_cost = torch.mean(ang_vel * ang_vel, dim=-1)
    ang_component = torch.tanh(ang_cost / angvel_temperature)
    fall_component = torch.clamp(0.2 - root_states[:, 2], min=0.0, max=1.0)

    # ---------------- total reward ----------------
    reward = (w_speed * speed_component
              + w_high_speed * high_speed_component
              + w_alive * alive_component
              - w_ctrl * ctrl_cost
              - w_angvel * ang_component
              - w_fall * fall_component)

    components: Dict[str, torch.Tensor] = {
        "forward_speed": speed_component,
        "high_speed": high_speed_component,
        "alive": alive_component,
        "ctrl_cost": ctrl_cost,
        "ang_vel_penalty": ang_component,
        "fall_penalty": fall_component,
    }
    return reward, components
